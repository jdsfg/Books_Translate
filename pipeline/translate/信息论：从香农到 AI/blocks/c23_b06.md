### D.7 逐层互信息探针

第 18 章 §18.4 报告了跨层的 MI 曲线实验。所用估计器是简单的离散代理：对隐藏激活分箱并在分箱变量上计算 MI。这对激活的低维投影是有效估计器，曲线的定性形状（低层保留原始字符身份；高层保留下一个 token 身份）是稳健的。

    def quantize(z, n_bins=64):
        """Hard-bin the first principal component of activations."""
        z_flat = z.reshape(-1, z.shape[-1])
        z_centered = z_flat - z_flat.mean(0, keepdim=True)
        u, s, v = torch.svd(z_centered)
        pc1 = (z_centered @ v[:, 0]).cpu().numpy()
        bins = torch.linspace(pc1.min(), pc1.max(), n_bins + 1)[1:-1].numpy()
        import numpy as np
        return np.digitize(pc1, bins)


    def mi_discrete(a, b):
        """Plug-in estimator of mutual information between discrete arrays."""
        import numpy as np
        from collections import Counter
        n = len(a)
        pa = Counter(a); pb = Counter(b); pab = Counter(zip(a, b))
        mi = 0.0
        for (ai, bi), c in pab.items():
            p_ab = c / n
            p_a = pa[ai] / n
            p_b = pb[bi] / n
            mi += p_ab * math.log2(p_ab / (p_a * p_b))
        return mi


    @torch.no_grad()
    def layer_mi_curve(model, sample_ids, ctx=256, batch=8):
        """For each layer, estimate I(layer_activation; input_char) and
        I(layer_activation; next_char)."""
        device = next(model.parameters()).device
        x = sample_ids[: batch * ctx].reshape(batch, ctx).to(device)
        y_input = x[:, -1].cpu().numpy()
        next_idx = batch * ctx
        y_next = sample_ids[next_idx : next_idx + batch].cpu().numpy()
        states = model.hidden_states(x)
        out = []
        for layer_idx, h in enumerate(states):
            h_last = h[:, -1, :]
            q = quantize(h_last)
            mi_in = mi_discrete(q, y_input)
            mi_next = mi_discrete(q, y_next)
            out.append((layer_idx, mi_in, mi_next))
        return out
