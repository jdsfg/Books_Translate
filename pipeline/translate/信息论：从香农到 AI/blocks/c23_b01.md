### D.2 数据加载


    DATA_URL = "https://raw.githubusercontent.com/karpathy/char-rnn/master/data/tinyshakespeare/input.txt"
    DATA_PATH = "tiny_shakespeare.txt"

    def fetch_data():
        if not os.path.exists(DATA_PATH):
            urllib.request.urlretrieve(DATA_URL, DATA_PATH)
        with open(DATA_PATH, "r", encoding="utf-8") as f:
            return f.read()

    text = fetch_data()
    n = len(text)
    n_train = int(0.9 * n)
    train_text, test_text = text[:n_train], text[n_train:]

    # Byte-level tokenization: each character is one token in [0, 256).
    def encode(s):
        return torch.tensor([b for b in s.encode("utf-8")], dtype=torch.long)

    train_ids = encode(train_text)
    test_ids = encode(test_text)

    class CharDataset(Dataset):
        def __init__(self, ids, ctx):
            self.ids = ids
            self.ctx = ctx
        def __len__(self):
            return len(self.ids) - self.ctx - 1
        def __getitem__(self, i):
            x = self.ids[i : i + self.ctx]
            y = self.ids[i + 1 : i + self.ctx + 1]
            return x, y
