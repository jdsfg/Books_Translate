### 8.7 TDD 的 mechanics：一个完整示例

考虑 Trellis 的一个真实形状问题：实现 `calculate_late_fee(invoice, current_date)` 函数。工程师理解的规格：未付余额的 5%，每逾期 30 天收一次，上限为发票总额的 25%。

TDD 周期 1：

    # Red
    def test_no_fee_when_not_late():
        invoice = Invoice(total=100, due_date=date(2026, 5, 1), paid=0)
        fee = calculate_late_fee(invoice, current_date=date(2026, 5, 1))
        assert fee == 0
    # 测试失败：函数不存在。

    # Green
    def calculate_late_fee(invoice, current_date):
        return 0
    # 测试通过。对该用例实现平凡正确。

    # Refactor：暂时无。

TDD 周期 2：

    # Red
    def test_fee_when_30_days_late():
        invoice = Invoice(total=100, due_date=date(2026, 5, 1), paid=0)
        fee = calculate_late_fee(invoice, current_date=date(2026, 5, 31))
        assert fee == 5  # 100 的 5%
    # 测试失败：函数返回 0。

    # Green
    def calculate_late_fee(invoice, current_date):
        days_late = (current_date - invoice.due_date).days
        if days_late < 30:
            return 0
        return invoice.total * 0.05
    # 测试通过。

    # Refactor：为清晰抽常量。
    LATE_FEE_RATE = 0.05
    def calculate_late_fee(invoice, current_date):
        days_late = (current_date - invoice.due_date).days
        if days_late < 30:
            return 0
        return invoice.total * LATE_FEE_RATE
    # 测试仍通过。

TDD 周期 3：

    # Red
    def test_fee_compounds_monthly():
        invoice = Invoice(total=100, due_date=date(2026, 5, 1), paid=0)
        fee = calculate_late_fee(invoice, current_date=date(2026, 7, 30))
        assert fee == 15  # 5% × 3（5、6、7 月周期）
    # 测试失败：函数返回 5。

    # Green
    def calculate_late_fee(invoice, current_date):
        days_late = (current_date - invoice.due_date).days
        if days_late < 30:
            return 0
        months_late = days_late // 30
        return invoice.total * LATE_FEE_RATE * months_late
    # 测试通过。

    # Refactor：暂时无。

TDD 周期 4：

    # Red
    def test_fee_capped_at_25_percent():
        invoice = Invoice(total=100, due_date=date(2026, 1, 1), paid=0)
        fee = calculate_late_fee(invoice, current_date=date(2026, 12, 31))
        assert fee == 25  # 364 天逾期 → 12 周期 × 5% = 60，上限为 100 的 25% = 25
    # 测试失败：函数返回 55。

    # Green
    def calculate_late_fee(invoice, current_date):
        days_late = (current_date - invoice.due_date).days
        if days_late < 30:
            return 0
        months_late = days_late // 30
        fee = invoice.total * LATE_FEE_RATE * months_late
        return min(fee, invoice.total * 0.25)
    # 测试通过。

    # Refactor：抽上限常量；考虑抽费用计算。
    LATE_FEE_RATE = 0.05
    LATE_FEE_CAP_RATE = 0.25

    def calculate_late_fee(invoice, current_date):
        days_late = (current_date - invoice.due_date).days
        if days_late < 30:
            return 0
        months_late = days_late // 30
        fee = invoice.total * LATE_FEE_RATE * months_late
        cap = invoice.total * LATE_FEE_CAP_RATE
        return min(fee, cap)
