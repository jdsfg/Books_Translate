        def release(self, sku: str, qty: int) -> None: ...

    # src/core/ports/audit_port.py
    class AuditPort(Protocol):
        def log(self, event: str, data: dict) -> None: ...

    # src/core/decisions/order_decision.py（纯函数）
    def decide_order(customer: Customer,
                     items: list[OrderItem],
                     current_date: date) -> OrderDecision:
        """纯决策：给定客户、商品和日期，应该发生什么？"""
        if customer.is_blocked:
            return OrderDecision(kind='reject', reason='customer_blocked')
        if not items:
            return OrderDecision(kind='reject', reason='empty_order')
        total = sum(item.price * item.quantity for item in items)
        discount = compute_discount(customer, total)
        final = total * (1 - discount)
        return OrderDecision(
            kind='accept',
            total=total,
            discount=discount,
            final_amount=final,
            items=items,
        )

    # src/core/orchestrators/order_orchestrator.py（命令式外壳）
    class OrderOrchestrator:
        def __init__(self, customers: CustomerPort, payment: PaymentPort,
                     inventory: InventoryPort, audit: AuditPort):
            self._customers = customers
            self._payment = payment
            self._inventory = inventory
            self._audit = audit

        def place_order(self, customer_id: str, items: list[OrderItem]) -> OrderResult:
            # 收集
            customer = self._customers.find(customer_id)
            if not customer:
                return OrderResult(status='error', error='customer_not_found')

            # 决策（纯）
            decision = decide_order(customer, items, current_date=date.today())

            # 动作
            if decision.kind == 'reject':
                self._audit.log('order_rejected', {'customer_id': customer_id, 'reason': decision.reason})
                return OrderResult(status='rejected', reason=decision.reason)

            for item in decision.items:
                if not self._inventory.reserve(item.sku, item.quantity):
                    # 回滚并拒绝
                    for prev_item in decision.items[:decision.items.index(item)]:  # 当前项索引
                        self._inventory.release(prev_item.sku, prev_item.quantity)
                    return OrderResult(status='rejected', reason='inventory_unavailable')

            charge_result = self._payment.charge(customer, decision.final_amount)
            if charge_result.status != 'success':
                for item in decision.items:
                    self._inventory.release(item.sku, item.quantity)
                return OrderResult(status='rejected', reason='payment_failed')

            self._audit.log('order_placed', {
                'customer_id': customer_id,
                'amount': str(decision.final_amount),
                'items': len(decision.items),
            })
            return OrderResult(status='success', total=decision.final_amount)
