### 7.7 完整示例：用 fake 的端到端测试设置

考虑 Helix 的 pricing 服务。它依赖 CustomerRepository、RuleEngine 和 EmailSender。完整 pricing 流程的测试：

    # tests/fakes/customer_repo.py
    class InMemoryCustomerRepository:
        def __init__(self): self._customers = {}
        def find(self, cid): return self._customers.get(cid)
        def save(self, c): self._customers[c.id] = c
        def find_by_region(self, region):
            return [c for c in self._customers.values() if c.region == region]

    # tests/fakes/email_sender.py
    class SpyEmailSender:
        def __init__(self): self.sent = []
        def send(self, to, subject, body):
            self.sent.append({'to': to, 'subject': subject, 'body': body})

    # tests/fakes/rule_engine.py
    class InMemoryRuleEngine:
        def __init__(self, rules): self._rules = rules
        def evaluate(self, customer, product):
            for rule in self._rules:
                if rule.matches(customer, product):
                    return rule.outcome
            return DEFAULT_PRICE

    # tests/pricing_service_test.py
    def test_enterprise_customer_gets_volume_discount_and_notification():
        # Arrange
        customers = InMemoryCustomerRepository()
        customers.save(Customer(id='c1', region='US', tier='enterprise'))
        rules = InMemoryRuleEngine([volume_discount_rule()])
        emails = SpyEmailSender()
        service = PricingService(customers, rules, emails)

        # Act
        result = service.price_for_customer('c1', product_id='p100', quantity=500)

        # Assert
        assert result.unit_price == pytest.approx(0.90)  # 10% 批量折扣
        assert result.total == pytest.approx(450.00)
        assert len(emails.sent) == 1
        assert 'volume discount' in emails.sent[0]['body']

测试读起来就是对行为的描述：高数量的企业客户获得折扣并被通知。没有 mock 设置。没有对内部调用结构的断言。重构 `PricingService` 如何调用依赖不会破坏测试，只要结果行为（价格和通知）被保留。

这就是本章主张的测试纪律。它在第一天比基于 mock 的版本有更多设置代码。三次重构后就能回本。
