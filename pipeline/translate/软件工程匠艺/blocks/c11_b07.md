        if order.refund_status == RefundStatus.FULL:
            raise InvalidStateError("Order already fully refunded")
        if request.amount <= 0 or request.amount != request.amount:  # NaN 防护
            raise ValueError(f"Invalid refund amount: {request.amount}")
        return ValidatedRequest(order=order, amount=request.amount)

    def _compute_refund_amount(order: Order, validated: ValidatedRequest) -> Decimal:
        remaining = order.total - order.total_refunded
        return min(Decimal(str(validated.amount)), remaining)

    def _build_result(order: Order, amount: Decimal) -> RefundResult:
        remaining_after = (order.total - order.total_refunded) - amount
        return RefundResult(
            order_id=order.id,
            amount=amount,
            currency=order.currency,
            status=RefundStatus.FULL if remaining_after == 0 else RefundStatus.PARTIAL,
